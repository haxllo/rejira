Create `apps/web/middleware.ts` (Next.js middleware — replaces `apps/web/proxy.ts`):

Per Next.js 16 convention, create a `middleware.ts` in `apps/web/` (not in `app/`):
```typescript
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";
import { auth } from "@/lib/auth/server";

const PUBLIC = [
  "/sign-in", "/sign-up", "/forgot-password", "/reset-password",
  "/verify-email", "/two-factor", "/check-email",
  "/api/auth", "/api/check", "/invite", "/",
];

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Allow public paths
  if (PUBLIC.some(p => pathname.startsWith(p))) {
    return NextResponse.next();
  }

  // Allow static assets, Next.js internals
  if (
    pathname.startsWith("/_next") ||
    pathname.startsWith("/static") ||
    pathname.includes(".")
  ) {
    return NextResponse.next();
  }

  // Check for Better Auth session cookie
  const sessionCookie =
    request.cookies.get("rejira.session_token")?.value ??
    request.cookies.get("__Secure-rejira.session_token")?.value;

  if (!sessionCookie) {
    const signInUrl = new URL("/sign-in", request.url);
    signInUrl.searchParams.set("next", pathname);
    return NextResponse.redirect(signInUrl);
  }

  // Session cookie exists — allow request through
  // (Better Auth validates the token server-side; middleware just gates)
  const response = NextResponse.next();

  // Set security headers
  response.headers.set("X-Frame-Options", "DENY");
  response.headers.set("X-Content-Type-Options", "nosniff");
  response.headers.set("Referrer-Policy", "strict-origin-when-cross-origin");

  return response;
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"],
};
```

Note: The middleware only checks cookie presence (not token validity) to avoid a DB hit on every request. Better Auth's route handlers and `requireAuth()` validate the actual token server-side.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → `/inbox`, `/my-issues`, `/settings` | All workspace routes now require valid session cookie; middleware gates unauthenticated |
| Client → `useSession()` | Client-side session state from Better Auth; token validated server-side on every API call |
| Middleware → route handler | Middleware checks cookie presence only (fast path); server actions/route handlers validate token |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-35 | Spoofing | Middleware bypass | mitigate | Middleware checks cookie presence; Better Auth validates token on all server-side operations even if middleware passes |
| T-03-36 | Elevation of Privilege | RequireAuth | mitigate | Server-side `requireAuth()` validates session + membership; client-side RequireAuth mirrors server check |
| T-03-37 | Information Disclosure | Old ME_ID code | mitigate | All `ME_ID` references deleted; no hardcoded user ID remains; RLS policies prevent cross-user data access |
| T-03-38 | Tampering | Session cookie | accept | Cookie is httpOnly, secure, sameSite=lax; theft risk mitigated in Plan 03-02 (IP/UA binding, new-device emails) |
</threat_model>
