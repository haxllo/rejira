```typescript
'use client';

import * as Y from 'yjs';
import { SupabaseProvider } from '@supabase-labs/y-supabase';
import { getSupabaseBrowserClient } from './client';
import type { SupabaseClient } from '@supabase/supabase-js';

/**
 * Room prefix for issue descriptions.
 * Room format: "issue:description:{issueExternalId}"
 */
const ROOM_PREFIX = 'issue:description:';

// Cache Y.Docs and providers so multiple editor instances share state
const docCache = new Map<string, { doc: Y.Doc; provider: SupabaseProvider; refCount: number }>();

/**
 * Get or create a Y.Doc and SupabaseProvider for collaborative editing.
 * Multiple callers for the same room share the same document.
 *
 * @returns Cleanup function that decrements the ref count
 */
export function createYjsProvider(
  issueExternalId: string,
  userInfo: { id: string; name: string; color: string },
): {
  doc: Y.Doc;
  yText: Y.Text;
  provider: SupabaseProvider;
  destroy: () => void;
} {
  const room = `${ROOM_PREFIX}${issueExternalId}`;
  const cached = docCache.get(room);
  
  if (cached) {
    cached.refCount++;
    return {
      doc: cached.doc,
      yText: cached.doc.getText('description'),
      provider: cached.provider,
      destroy: () => {
        cached.refCount--;
        if (cached.refCount <= 0) {
          cached.provider.destroy();
          cached.doc.destroy();
          docCache.delete(room);
        }
      },
    };
  }

  const doc = new Y.Doc();
  const supabase = getSupabaseBrowserClient();
  const provider = new SupabaseProvider(room, doc, supabase, {
    awareness: true,
    persistence: true,
  });

  // Set local awareness state (cursor presence)
  const awareness = provider.getAwareness();
  if (awareness) {
    awareness.setLocalStateField('user', {
      id: userInfo.id,
      name: userInfo.name,
      color: userInfo.color,
    });
  }

  const entry = { doc, provider, refCount: 1 };
  docCache.set(room, entry);

  return {
    doc,
    yText: doc.getText('description'),
    provider,
    destroy: () => {
      entry.refCount--;
      if (entry.refCount <= 0) {
        provider.destroy();
        doc.destroy();
        docCache.delete(room);
      }
    },
  };
}

/**
 * Get the awareness state from a room to show who's currently editing.
 * Returns null if the room has no active provider.
 */
export function getYjsAwarenessState(issueExternalId: string): Array<{
  id: string;
  name: string;
  color: string;
}> | null {
  const room = `${ROOM_PREFIX}${issueExternalId}`;
  const cached = docCache.get(room);
  if (!cached) return null;

  const awareness = cached.provider.getAwareness();
  if (!awareness) return null;

  const states: Array<{ id: string; name: string; color: string }> = [];
  for (const [, state] of awareness.getStates()) {
    const user = (state as Record<string, unknown>)?.user as { id: string; name: string; color: string } | undefined;
    if (user) states.push(user);
  }
  return states;
}
```

CREATE TABLE IF NOT EXISTS public.yjs_documents (
  room TEXT PRIMARY KEY,
  state TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

ALTER TABLE public.yjs_documents ENABLE ROW LEVEL SECURITY;

-- Members in the same workspace can read/write Yjs documents
-- The workspace is derived from the room name prefix (issue:description:{wsId}:{issueId})
CREATE POLICY "workspace_members_can_access_yjs_docs"
  ON public.yjs_documents
  FOR ALL
  USING (
    SPLIT_PART(room, ':', 3)::uuid IN (
      SELECT workspace_id FROM memberships WHERE user_id = auth.uid()::text
    )
  )
  WITH CHECK (
    SPLIT_PART(room, ':', 3)::uuid IN (
      SELECT workspace_id FROM memberships WHERE user_id = auth.uid()::text
    )
  );
```

'use client';

import { useEffect, useRef, useState, useCallback } from 'react';
import { createYjsProvider } from '@/lib/realtime/yjs-provider';
import { useUser } from '@/hooks/useUser';
import type * as Y from 'yjs';

interface IssueDescriptionEditorProps {
  issueExternalId: string;
  initialDescription: string;
  onSave?: (text: string) => void;
  readOnly?: boolean;
}

export function IssueDescriptionEditor({
  issueExternalId,
  initialDescription,
  onSave,
  readOnly = false,
}: IssueDescriptionEditorProps) {
  const { user } = useUser();
  const textareaRef = useRef<HTMLTextAreaElement>(null);
  const [yText, setYText] = useState<Y.Text | null>(null);
  const [value, setValue] = useState(initialDescription);
  const [isCollaborative, setIsCollaborative] = useState(false);
  const [coEditors, setCoEditors] = useState<string[]>([]);
  const yTextRef = useRef<Y.Text | null>(null);
  const isInternalUpdate = useRef(false);

  useEffect(() => {
    if (!user?.id || !issueExternalId) return;

    const { yText: yt, destroy } = createYjsProvider(
      issueExternalId,
      { id: user.id, name: user.name ?? 'Anonymous', color: user.avatarColor ?? '#6366f1' },
    );

    yTextRef.current = yt;
    setYText(yt);
    setIsCollaborative(true);

    // Sync Yjs text → local state
    const observer = () => {
      isInternalUpdate.current = true;
      setValue(yt.toString());
    };
    yt.observe(observer);

    return () => {
      yt.unobserve(observer);
      destroy();
    };
  }, [issueExternalId, user?.id, user?.name, user?.avatarColor]);

  // Handle local edits → Yjs
  const handleChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
    if (isInternalUpdate.current) {
      isInternalUpdate.current = false;
      return;
    }
    const newValue = e.target.value;
    setValue(newValue);
    
    const yt = yTextRef.current;
    if (yt) {
      // Apply the diff to Yjs (simple approach: replace entire content)
      // For production, use y-prosemirror with a rich text editor
      yt.delete(0, yt.length);
      yt.insert(0, newValue);
    }
  }, []);

  return (
    <div className="relative">
      {isCollaborative && coEditors.length > 0 && (
        <div className="mb-1 flex items-center gap-1 text-[10px] text-[var(--color-text-muted)]">
          <span className="size-1.5 rounded-full bg-[var(--color-accent)]" />
          {coEditors.join(', ')} {coEditors.length === 1 ? 'is' : 'are'} editing
        </div>
      )}
      <textarea
        ref={textareaRef}
        value={value}
        onChange={handleChange}
        readOnly={readOnly}
        className="w-full resize-none rounded-md border border-[var(--color-border)] bg-[var(--color-surface-1)] p-3 text-[13px] text-[var(--color-text)] outline-none transition-colors focus:border-[var(--color-accent)] focus:ring-1 focus:ring-[var(--color-accent)]"
        rows={6}
        placeholder="Add a description..."
      />
    </div>
  );
}
```

If the existing `issue-description-editor.tsx` already has rich editing (ProseMirror/TipTap), use `y-prosemirror` instead of a textarea binding.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Yjs document ↔ Broadcast channel | Peer-to-peer: Yjs syncs binary state over Supabase Realtime Broadcast. All clients in the same room receive all updates. |
| Yjs awareness ↔ Broadcast channel | Peer-to-peer: awareness (cursor, user info) is broadcast to all room members |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-16 | Spoofing | Yjs awareness | accept | Awareness state carries userId/name which clients set themselves. This is a UX feature, not an auth boundary. No action is taken based on awareness state. |
| T-05-17 | Information Disclosure | Yjs document content via Broadcast | mitigate | Yjs syncs binary content over the Supabase Realtime Broadcast channel. Only clients subscribed to the specific room can see updates. Room names include the issue ID and workspace ID, making accidental cross-workspace leakage impossible. RLS on the workspace provider channel prevents unauthorized subscription. |
| T-05-18 | Tampering | Yjs document state | accept | Any client in the room can modify the Yjs document. Since all changes are CRDT-merged, malicious edits can't corrupt state beyond what the user could do via the UI. Yjs persistence writes the merged state back to Postgres. |
</threat_model>
