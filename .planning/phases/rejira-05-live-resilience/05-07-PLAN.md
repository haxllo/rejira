// At the top of the component, add state:

import { useState } from 'react';
import { MenuIcon, XIcon } from '@/components/icons';

// Inside the component:
const [mobileOpen, setMobileOpen] = useState(false);

<aside className={cn(
  "flex flex-col border-r border-[var(--color-border)] bg-[var(--color-bg)]",

  // Mobile: overlay drawer
  mobileOpen ? "fixed inset-0 z-50 flex w-[280px]" : "hidden",

// Hamburger button for mobile (shown outside the nav):
<button
  onClick={() => setMobileOpen(true)}
  className="fixed bottom-4 left-4 z-40 grid size-10 place-items-center rounded-full bg-[var(--color-surface-2)] shadow-lg md:hidden"

</button>

/* Safe area padding for notched devices */
@supports (padding: env(safe-area-inset-bottom)) {
