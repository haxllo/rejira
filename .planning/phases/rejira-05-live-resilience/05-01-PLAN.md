```typescript
'use client';

import { useEffect, useSyncExternalStore } from 'react';
import { getSupabaseBrowserClient } from './client';
import type { RealtimeChannel } from '@supabase/supabase-js';

export interface PresenceState {
  userId: string;
  name: string;
  avatarColor: string;
  currentIssueId: string | null;
  onlineAt: string;
}

// Module-level state shared across components
let presenceState: Map<string, PresenceState> = new Map();
let listeners: Set<() => void> = new Set();

function notifyListeners() {
  listeners.forEach((l) => l());
}

function subscribeToPresenceChanges(listener: () => void): () => void {
  listeners.add(listener);
  return () => listeners.delete(listener);
}

function getPresenceSnapshot(): Map<string, PresenceState> {
  return presenceState;
}

/**
 * Call inside the workspace provider's effect to start tracking presence.
 * Returns an unsubscribe function.
 */
export function startPresenceTracking(
  channel: RealtimeChannel,
  userId: string,
  name: string,
  avatarColor: string,
  getCurrentIssueId: () => string | null,
): () => void {
  let destroyed = false;

  channel.on('presence', { event: 'sync' }, () => {
    if (destroyed) return;
    const state = channel.presenceState() as Record<string, Record<string, unknown>[]>;
    const newMap = new Map<string, PresenceState>();
    for (const [key, entries] of Object.entries(state)) {
      const entry = entries[0] as Record<string, unknown> | undefined;
      if (entry && entry.userId !== userId) {
        // Don't include self
        newMap.set(key, entry as unknown as PresenceState);
      }
    }
    presenceState = newMap;
    notifyListeners();
  });

  // Track own presence
  const track = () => {
    if (destroyed) return;
    channel.track({
      userId,
      name,
      avatarColor,
      currentIssueId: getCurrentIssueId(),
      onlineAt: new Date().toISOString(),
    });
  };

  // Track immediately and set up interval
  track();
  const interval = setInterval(track, 30_000);

  return () => {
    destroyed = true;
    clearInterval(interval);
    // Presence is automatically cleaned up by Supabase on channel leave
  };
}

/**
 * Hook to subscribe to presence changes.
 */
export function usePresence(): Map<string, PresenceState> {
  return useSyncExternalStore(subscribeToPresenceChanges, getPresenceSnapshot, getPresenceSnapshot);
}

/**
 * Get presence for a specific issue.
 */
export function useIssuePresence(issueId: string): PresenceState[] {
  const all = usePresence();
  return Array.from(all.values()).filter((p) => p.currentIssueId === issueId);
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → Supabase Realtime | Untrusted: any connected client can send presence data; we MUST sanitize display (no HTML injection in presence names — they render via style/attribute bindings which React escapes) |
| Realtime callback → zustand | Trusted: Supabase Realtime Postgres Changes payloads are JSON from our own DB; RLS enforces workspace boundaries before the change is published |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-01 | Spoofing | presence.track() | accept | Presence is peer-to-peer across connected clients; we filter out self but trust other clients' userId claim. Accepting because presence is non-authoritative (no action based on it). |
| T-05-02 | Information Disclosure | presence state | mitigate | Presence only shares userId, name, avatarColor, currentIssueId — no email, no session tokens, no private data. |
| T-05-03 | Denial of Service | presence tracking | accept | A malicious client could flood presence sync events; Supabase Realtime has built-in rate limiting at the channel level. |
| T-05-04 | Tampering | Realtime payload → setOne | mitigate | We trust Realtime Postgres Changes payloads because they originate from our own DB behind RLS. Risk: if RLS is misconfigured, a user could see cross-workspace data via presence leaks. RLS is verified in Phase 2 tests. |
</threat_model>
