   ```typescript
   function describe(activity: ActivityWithActor): string {
     const verb = VERB_LABEL[activity.verb] ?? activity.verb.replace(/_/g, ' ');
     const before = (activity.before ?? {}) as Record<string, unknown>;
     const after = (activity.after ?? {}) as Record<string, unknown>;
     const metadata = { ...before, ...after } as Record<string, unknown>;
     const objectType = activity.objectType.replace(/s$/, '');

     if (objectType === 'issue') {
       const key = after.key ?? before.key ?? '';
       const title = after.title ?? before.title ?? '';
       const base = key ? `${verb} ${key}${title ? ` (${title})` : ''}` : `${verb} issue #${activity.objectId}`;

       // Add diff details for specific verbs
       if (activity.verb === 'status_changed' && before.status && after.status) {
         return `${base} — ${String(before.status)} → ${String(after.status)}`;
       }
       if (activity.verb === 'assigned' && after.assignee) {
         return `${base} to ${String(after.assignee)}`;
       }
       if (activity.verb === 'updated') {
         const changedFields = Object.keys(after).filter(k => k !== 'updatedAt' && k !== 'sortOrder');
         if (changedFields.length > 0) {
           return `${base} — ${changedFields.join(', ')}`;
         }
       }
       return base;
     }

     if (objectType === 'project' && metadata.name) {
       return `${verb} ${String(metadata.name)}`;
     }
     if (objectType === 'cycle' && metadata.name) {
       return `${verb} ${String(metadata.name)}`;
     }
     return `${verb} ${objectType} #${activity.objectId}`;
   }
   ```
