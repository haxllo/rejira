<action>
Update `apps/web/lib/ai/cost.ts` to implement `recordAICost`:

```typescript
import { db } from '@/lib/db';
import { workspaceSecurityPolicy } from '@/lib/db/schema';
import { eq, sql } from 'drizzle-orm';

export async function recordAICost(
  workspaceId: string,
  model: string,
  tokens: number
): Promise<void> {
  // Pricing per 1K tokens (approximate)
  const rates: Record<string, number> = {
    'gpt-4o-mini': 0.00015, // $0.15/M input tokens
    'gpt-4o': 0.0025,
    'text-embedding-3-small': 0.00002,
  };
  const ratePerToken = rates[model] ?? 0.00015;
  const costCents = Math.ceil(tokens * ratePerToken * 100);

  await db
    .update(workspaceSecurityPolicy)
    .set({
      monthlyAiSpentCents: sql`${workspaceSecurityPolicy.monthlyAiSpentCents} + ${costCents}`,
    })
    .where(eq(workspaceSecurityPolicy.workspaceId, workspaceId));
}
```

Create a budget check helper:

```typescript
export async function checkAIBudget(workspaceId: string): Promise<{ withinBudget: boolean; spent: number; budget: number }> {
  const policy = await db
    .select({
      spent: workspaceSecurityPolicy.monthlyAiSpentCents,
      budget: workspaceSecurityPolicy.monthlyAiBudgetCents,
    })
    .from(workspaceSecurityPolicy)
    .where(eq(workspaceSecurityPolicy.workspaceId, workspaceId))
    .then(r => r[0]);

  if (!policy) return { withinBudget: true, spent: 0, budget: 0 };
  return {
    withinBudget: (policy.spent ?? 0) < (policy.budget ?? 1000),
    spent: policy.spent ?? 0,
    budget: policy.budget ?? 1000,
  };
}
```

Wire `checkAIBudget` into the triage and summarize route handlers — return 429 with `{ error: 'Monthly AI budget exceeded', spent, budget }` if over budget.
</action>
