```typescript
import type { NextConfig } from "next";

const nextConfig: NextConfig = {
  reactStrictMode: true,
  
  // Compress responses
  compress: true,

  // Production browser source maps disabled (smaller bundles)
  productionBrowserSourceMaps: false,

  // Optimize package imports (already configured + add more)
  experimental: {
    optimizePackageImports: [
      "motion",
      "@radix-ui/react-dialog",
      "@radix-ui/react-popover",
      "@radix-ui/react-dropdown-menu",
      "@radix-ui/react-tooltip",
      "cmdk",
      "date-fns",
      "@dnd-kit/core",
      "@dnd-kit/sortable",
    ],
  },

  // Image optimization (use next/image for remote images)
  images: {
    formats: ["image/avif", "image/webp"],
    deviceSizes: [640, 750, 828, 1080, 1200, 1920],
    imageSizes: [16, 32, 48, 64, 96, 128, 256],
    minimumCacheTTL: 31536000, // 1 year for static images
    remotePatterns: [
      {
        protocol: "https",
        hostname: "*.supabase.co",
        pathname: "/storage/v1/object/public/**",
      },
    ],
  },

  // Cache-Control headers for static assets
  async headers() {
    return [
      // ... existing security headers from Phase 3

      // Static assets: immutable cache
      {
        source: "/_next/static/(.*)",
        headers: [
          { key: "Cache-Control", value: "public, max-age=31536000, immutable" },
        ],
      },

      // Images: long cache
      {
        source: "/images/(.*)",
        headers: [
          { key: "Cache-Control", value: "public, max-age=86400, stale-while-revalidate=604800" },
        ],
      },

      // Fonts: long cache
      {
        source: "/fonts/(.*)",
        headers: [
          { key: "Cache-Control", value: "public, max-age=31536000, immutable" },
        ],
      },

      // HTML pages: short cache with revalidation
      {
        source: "/((?!_next|api|images|fonts).*)",
        headers: [
          { key: "Cache-Control", value: "public, max-age=0, must-revalidate" },
        ],
      },
    ];
  },

  // Redirects for trailing slashes, www, etc.
  async redirects() {
    return [
      { source: "/index", destination: "/", permanent: true },
    ];
  },
};

export default nextConfig;
```

Create `e2e/perf/lighthouse.spec.ts` — Playwright + Lighthouse audit:

```typescript
import { test, expect } from '@playwright/test';
import { playAudit } from 'playwright-lighthouse';
import lighthouse from 'lighthouse';

const ROUTES = [
  { name: 'Inbox', path: '/inbox' },
  { name: 'My Issues', path: '/my-issues' },
  { name: 'Project Issues', path: '/projects/eng/issues' },
  { name: 'Cycle Board', path: '/projects/eng/cycles/1' },
  { name: 'Settings', path: '/settings' },
  { name: 'Sign In', path: '/sign-in' },
];

const THRESHOLDS = {
  performance: 95,
  accessibility: 95,
  'best-practices': 95,
  seo: 95,
};

for (const route of ROUTES) {
  test(`Lighthouse: ${route.name} > 95`, async ({ page, browserName }) => {
    // Only run on Chromium (Lighthouse requires it)
    if (browserName !== 'chromium') {
      test.skip();
    }
    
    await page.goto(route.path, { waitUntil: 'networkidle' });
    
    await playAudit({
      page,
      port: 9222,
      thresholds: THRESHOLDS,
      reports: {
        formats: { html: true },
        name: `lighthouse-${route.name.toLowerCase().replace(/\s+/g, '-')}`,
        directory: 'lighthouse-reports',
      },
    });
  });
}

// Core Web Vitals specific tests
test('LCP < 1.2s on Inbox', async ({ page }) => {
  await page.goto('/inbox');
  const lcp = await page.evaluate(() => {
    return new Promise<number>((resolve) => {
      new PerformanceObserver((list) => {
        const entries = list.getEntries();
        const lastEntry = entries[entries.length - 1];
        resolve(lastEntry.startTime);
      }).observe({ type: 'largest-contentful-paint', buffered: true });
      setTimeout(() => resolve(0), 10000); // timeout
    });
  });
  expect(lcp).toBeLessThan(1200);
});

test('CLS < 0.05 on Inbox', async ({ page }) => {
  await page.goto('/inbox');
  await page.waitForLoadState('networkidle');
  const cls = await page.evaluate(() => {
    return new Promise<number>((resolve) => {
      let cumulativeScore = 0;
      new PerformanceObserver((list) => {
        for (const entry of list.getEntries()) {
          if (!(entry as any).hadRecentInput) {
            cumulativeScore += (entry as any).value;
          }
        }
      }).observe({ type: 'layout-shift', buffered: true });
      setTimeout(() => resolve(cumulativeScore), 3000);
    });
  });
  expect(cls).toBeLessThan(0.05);
});

test('INP < 200ms on interactive elements', async ({ page }) => {
  await page.goto('/inbox');
  await page.waitForLoadState('networkidle');
  
  // Click first issue row and measure response
  const start = Date.now();
  await page.click('[data-testid="issue-row"]:first-child');
  await page.waitForSelector('[data-testid="issue-drawer"]');
  const inp = Date.now() - start;
  expect(inp).toBeLessThan(200);
});
```

**Common performance bottlenecks to fix:**

1. **Large JS bundles** — Check bundle analyzer output. Lazy-load heavy components with `next/dynamic`:
   ```tsx
   const IssueDrawer = dynamic(() => import('@/components/issue/issue-drawer'), {
     ssr: false,
     loading: () => <DrawerSkeleton />,
   });
   const CommandPalette = dynamic(() => import('@/components/shell/command-palette'), {
     ssr: false,
   });
   ```

2. **Unoptimized images** — Replace any `<img>` tags displaying Supabase Storage images with `next/image`:
   ```tsx
   import Image from 'next/image';
   <Image src={supabaseUrl} alt={user.name} width={32} height={32} />
   ```
   For avatar images specifically, add explicit width/height to prevent CLS.

3. **Render-blocking resources** — The Google Fonts stylesheet is render-blocking. Mitigation:
   - Already using `display=swap` (task 1)
   - Consider self-hosting fonts via `next/font/google` instead of `<link>` for optimal loading
   - Use `next/font/google` for Geist:
   ```tsx
   import { Geist, Geist_Mono } from 'next/font/google';
   const geist = Geist({ subsets: ['latin'], display: 'swap' });
   const geistMono = Geist_Mono({ subsets: ['latin'], display: 'swap' });
   ```

4. **Large component trees** — Audit `IssueRow`, `GroupedList`, `IssueDrawer` for unnecessary re-renders. Memoize expensive computations with `useMemo`. Use `React.memo` for list items.

5. **Third-party scripts** — PostHog and Sentry are loaded in Phase 3. Ensure they use `next/script` with `strategy="lazyOnload"` or `strategy="afterInteractive"`:
   ```tsx
   <Script src="https://app.posthog.com/..." strategy="lazyOnload" />
   ```

6. **Tailwind purge** — Tailwind v4 automatically tree-shakes unused classes. Verify the production build doesn't include unused CSS.
