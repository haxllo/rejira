Create `apps/web/components/legal/cookie-consent.tsx` ('use client'):
```tsx
'use client';

import { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'motion/react';

export function CookieConsent() {
  const [show, setShow] = useState(false);

  useEffect(() => {
    const consented = localStorage.getItem('cookie-consent');
    if (!consented) setShow(true);
  }, []);

  function acceptAll() {
    localStorage.setItem('cookie-consent', 'all');
    setShow(false);
    // Enable PostHog, analytics cookies
    window.dispatchEvent(new CustomEvent('cookies-accepted'));
  }

  function acceptEssential() {
    localStorage.setItem('cookie-consent', 'essential');
    setShow(false);
    // Only session/auth cookies active
  }

  if (!show) return null;

  return (
    <AnimatePresence>
      <motion.div
        initial={{ y: 100, opacity: 0 }}
        animate={{ y: 0, opacity: 1 }}
        exit={{ y: 100, opacity: 0 }}
        className="fixed bottom-4 left-4 right-4 z-50 p-4 bg-[var(--color-surface-2)] border border-[var(--color-border)] rounded-lg shadow-lg max-w-lg mx-auto"
        data-testid="cookie-consent"
      >
        <p className="text-sm text-[var(--color-text)] mb-3">
          We use cookies to improve your experience. Essential cookies are required for the app to function. Analytics cookies help us understand usage.
        </p>
        <div className="flex gap-2 flex-wrap">
          <button onClick={acceptAll} className="px-4 py-2 bg-[var(--color-accent)] text-white rounded text-sm font-medium">
            Accept All
          </button>
          <button onClick={acceptEssential} className="px-4 py-2 bg-[var(--color-surface-3)] text-[var(--color-text)] rounded text-sm">
            Essential Only
          </button>
          <a href="/legal/privacy" className="px-4 py-2 text-sm text-[var(--color-text-muted)] underline self-center">
            Privacy Policy
          </a>
        </div>
      </motion.div>
    </AnimatePresence>
  );
}
```
