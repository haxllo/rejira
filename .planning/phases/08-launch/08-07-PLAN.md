Create `apps/web/lib/i18n/date-time.ts`:
```typescript
import type { Locale } from './dict';

const LOCALE_MAP: Record<Locale, string> = {
  en: 'en-US',
  es: 'es-ES',
  fr: 'fr-FR',
  de: 'de-DE',
  ja: 'ja-JP',
  zh: 'zh-CN',
};

export function formatDate(
  date: Date | string,
  locale: Locale,
  timezone?: string,
  style: 'short' | 'medium' | 'long' | 'full' = 'medium',
): string {
  const d = typeof date === 'string' ? new Date(date) : date;
  const localeStr = LOCALE_MAP[locale];
  return new Intl.DateTimeFormat(localeStr, {
    dateStyle: style,
    timeZone: timezone || undefined,
  }).format(d);
}

export function formatDateTime(
  date: Date | string,
  locale: Locale,
  timezone?: string,
): string {
  const d = typeof date === 'string' ? new Date(date) : date;
  const localeStr = LOCALE_MAP[locale];
  return new Intl.DateTimeFormat(localeStr, {
    dateStyle: 'medium',
    timeStyle: 'short',
    timeZone: timezone || undefined,
  }).format(d);
}

export function formatRelativeTime(
  date: Date | string,
  locale: Locale,
): string {
  const d = typeof date === 'string' ? new Date(date) : date;
  const now = new Date();
  const diffMs = now.getTime() - d.getTime();
  const diffSec = Math.floor(diffMs / 1000);
  const diffMin = Math.floor(diffSec / 60);
  const diffHour = Math.floor(diffMin / 60);
  const diffDay = Math.floor(diffHour / 24);

  const rtf = new Intl.RelativeTimeFormat(LOCALE_MAP[locale], { numeric: 'auto' });

  if (diffSec < 60) return rtf.format(-diffSec, 'second');
  if (diffMin < 60) return rtf.format(-diffMin, 'minute');
  if (diffHour < 24) return rtf.format(-diffHour, 'hour');
  if (diffDay < 7) return rtf.format(-diffDay, 'day');
  if (diffDay < 30) return rtf.format(-Math.floor(diffDay / 7), 'week');
  if (diffDay < 365) return rtf.format(-Math.floor(diffDay / 30), 'month');
  return rtf.format(-Math.floor(diffDay / 365), 'year');
}

export function getUserTimezone(): string {
  return Intl.DateTimeFormat().resolvedOptions().timeZone;
}
```

Create `apps/web/components/primitives/date-display.tsx` ('use client'):
```tsx
'use client';

import { useLocale } from '@/hooks/useLocale';
import { formatDate, formatRelativeTime, formatDateTime } from '@/lib/i18n/date-time';
import { useState, useEffect } from 'react';

type DateDisplayProps = {
  date: string | Date;
  variant?: 'relative' | 'short' | 'full';
  className?: string;
};

export function DateDisplay({ date, variant = 'relative', className }: DateDisplayProps) {
  const { locale } = useLocale();
  const [display, setDisplay] = useState('');

  useEffect(() => {
    if (variant === 'relative') {
      setDisplay(formatRelativeTime(date, locale));
    } else if (variant === 'short') {
      setDisplay(formatDate(date, locale, undefined, 'short'));
    } else {
      setDisplay(formatDateTime(date, locale));
    }
  }, [date, locale, variant]);

  return (
    <time dateTime={typeof date === 'string' ? date : date.toISOString()} className={className}>
      {display}
    </time>
  );
}
```
