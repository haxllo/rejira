Create `e2e/a11y/all-screens.spec.ts` — Playwright + axe-core audit for all 6 core workspace screens:

```typescript
import { test, expect } from '@playwright/test';
import { injectAxe, checkA11y, getViolations } from '@axe-core/playwright';

const SCREENS = [
  { name: 'Inbox', url: '/inbox', selector: '[data-testid="inbox-page"]' },
  { name: 'My Issues', url: '/my-issues', selector: '[data-testid="my-issues-page"]' },
  { name: 'Project Issues', url: '/projects/eng/issues', selector: '[data-testid="project-issues-page"]' },
  { name: 'Cycle Board', url: '/projects/eng/cycles/1', selector: '[data-testid="cycle-board-page"]' },
  { name: 'Settings', url: '/settings', selector: '[data-testid="settings-page"]' },
];

for (const screen of SCREENS) {
  test(`WCAG 2.2 AA: ${screen.name} page has 0 critical/serious violations`, async ({ page }) => {
    await page.goto(screen.url);
    await page.waitForSelector(screen.selector, { timeout: 10000 });
    await injectAxe(page);
    const violations = await checkA11y(page, screen.selector, {
      detailedReport: true,
      detailedReportOptions: { html: true },
    });
    const critical = violations.filter(v => v.impact === 'critical');
    const serious = violations.filter(v => v.impact === 'serious');
    expect(critical).toEqual([]);
    expect(serious).toEqual([]);
  });
}

// Issue Drawer test (overlay — must be opened first)
test('WCAG 2.2 AA: Issue Drawer has 0 critical/serious violations', async ({ page }) => {
  await page.goto('/my-issues');
  await page.waitForSelector('[data-testid="my-issues-page"]');
  // Open first issue in drawer
  await page.click('[data-testid="issue-row"]:first-child');
  await page.waitForSelector('[data-testid="issue-drawer"]');
  await injectAxe(page);
  const violations = await checkA11y(page, '[data-testid="issue-drawer"]', {
    detailedReport: true,
    detailedReportOptions: { html: true },
  });
  const critical = violations.filter(v => v.impact === 'critical');
  const serious = violations.filter(v => v.impact === 'serious');
  expect(critical).toEqual([]);
  expect(serious).toEqual([]);
});

// Color contrast audit (separate — uses axe-core color contrast rule specifically)
test('Color contrast passes WCAG AA minimum on all core screens', async ({ page }) => {
  for (const screen of SCREENS) {
    await page.goto(screen.url);
    await page.waitForSelector(screen.selector, { timeout: 10000 });
    await injectAxe(page);
    const violations = await checkA11y(page, screen.selector, {
      runOnly: ['color-contrast'],
    });
    expect(violations).toEqual([]);
  }
});
```

Create `e2e/a11y/keyboard-full.spec.ts` — full keyboard navigation audit:

```typescript
import { test, expect } from '@playwright/test';

test('Keyboard navigation: Inbox — Tab through all interactive elements', async ({ page }) => {
  await page.goto('/inbox');
  await page.waitForSelector('[data-testid="inbox-page"]');
  
  // Start focus on body, then Tab through
  await page.keyboard.press('Tab');
  // PrimaryNav items
  await expect(page.locator(':focus')).toBeVisible();
  // Continue Tab through all items, count total focusable elements
  let tabCount = 0;
  let previousFocused = '';
  for (let i = 0; i < 50; i++) {
    await page.keyboard.press('Tab');
    const focused = await page.evaluate(() => document.activeElement?.tagName || '');
    if (focused === previousFocused) break; // wrapped around
    previousFocused = focused;
    tabCount++;
  }
  expect(tabCount).toBeGreaterThan(5); // at least some focusable elements exist
  
  // Focus ring must be visible
  const hasFocusRing = await page.evaluate(() => {
    const el = document.activeElement;
    if (!el) return false;
    const style = window.getComputedStyle(el);
    return style.outlineWidth !== '0px' && style.outlineStyle !== 'none';
  });
  // Note: focus ring only visible after keyboard interaction
});

test('Keyboard: Escape dismisses drawer and command palette', async ({ page }) => {
  await page.goto('/my-issues');
  // Open drawer
  await page.click('[data-testid="issue-row"]:first-child');
  await page.waitForSelector('[data-testid="issue-drawer"]');
  // Escape closes
  await page.keyboard.press('Escape');
  await expect(page.locator('[data-testid="issue-drawer"]')).not.toBeVisible();
  
  // Open command palette
  await page.keyboard.press('Meta+k');
  await page.waitForSelector('[data-testid="command-palette"]');
  await page.keyboard.press('Escape');
  await expect(page.locator('[data-testid="command-palette"]')).not.toBeVisible();
});

test('Keyboard: Status change via keyboard (1-5) while row is focused', async ({ page }) => {
  await page.goto('/my-issues');
  // Focus first issue row
  await page.click('[data-testid="issue-row"]:first-child [data-testid="status-selector"]');
  // Press 3 for "In Progress"
  await page.keyboard.press('3');
  // Status should have changed
});
```

Run the axe-core audit from task 1 and fix ALL violations found. Common issues and their fixes in a Next.js/Tailwind/Radix app:

1. **Missing form labels** — Every `<input>`, `<select>`, `<textarea>` must have an associated `<label>` with `htmlFor` or `aria-label`. Check all auth forms, filter inputs, search inputs, settings forms.

2. **Missing alt text on images** — Every `<img>` must have `alt` attribute. Decorative images: `alt=""`. Avatars: `alt={user.name}`. Icons with text labels: `alt=""` (the text provides the meaning).

3. **Insufficient color contrast** — Check OKLCH tokens in `apps/web/app/globals.css`. Text on surfaces must meet 4.5:1. Add `--color-text-on-surface-1`, `--color-text-muted-on-surface-2` tokens if needed. The dark-mode-first approach means light text on dark surfaces — verify light mode tokens meet contrast too.

4. **Focus order / tabindex issues** — Remove positive `tabindex` values. Use `tabindex="0"` only for non-interactive elements that need focus. `tabindex="-1"` for programmatic focus. Radix UI dialogs/popovers handle focus trapping — verify it works.

5. **ARIA roles and states** — Radix UI primitives handle ARIA correctly, but custom components may need explicit roles. Verify: `role="button"` on clickable non-button elements, `aria-expanded` on toggles, `aria-label` on icon-only buttons.

6. **Skip link** — Add a skip-to-main-content link as the first focusable element:
```tsx
// In apps/web/app/(workspace)/layout.tsx
<a href="#main-content" className="sr-only focus:not-sr-only focus:absolute focus:top-2 focus:left-2 focus:z-50 focus:px-4 focus:py-2 focus:bg-[var(--color-surface-1)] focus:border focus:border-[var(--color-border-focus)] focus:rounded">
  Skip to main content
</a>
```

7. **Live regions for dynamic content** — Add `aria-live="polite"` to areas that update dynamically: toast container, notification badge, bulk action bar. Add `aria-live="assertive"` for error messages.

8. **Motion sensitivity** — Verify the existing `prefers-reduced-motion` block in `globals.css` works. Test by enabling "Reduce motion" in OS settings.

9. **Touch targets** — Mobile: interactive elements must be ≥ 44×44px. Check density modes — compact mode may shrink targets below minimum.

10. **Heading hierarchy** — Each page should have one `<h1>`, headings should not skip levels. Verify page structure.
