```typescript
import 'server-only';
import Stripe from 'stripe';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, {
  apiVersion: '2025-03-31.basil' as any, // Latest stable
  typescript: true,
});

const PRICE_IDS: Record<string, string> = {
  pro: process.env.STRIPE_PRO_PRICE_ID!,
  enterprise: process.env.STRIPE_ENTERPRISE_PRICE_ID!,
};

export async function createCheckoutSession(params: {
  workspaceId: string;
  workspaceSlug: string;
  tier: 'pro' | 'enterprise';
  customerEmail: string;
}): Promise<{ url: string }> {
  const { workspaceId, workspaceSlug, tier, customerEmail } = params;

  const session = await stripe.checkout.sessions.create({
    mode: 'subscription',
    customer_email: customerEmail,
    line_items: [{ price: PRICE_IDS[tier], quantity: 1 }],
    metadata: { workspaceId, tier },
    success_url: `${process.env.BETTER_AUTH_URL}/settings/billing?success=true`,
    cancel_url: `${process.env.BETTER_AUTH_URL}/settings/billing?canceled=true`,
    subscription_data: {
      metadata: { workspaceId, tier },
    },
    allow_promotion_codes: true,
    billing_address_collection: 'auto',
  });

  return { url: session.url! };
}

export async function createCustomerPortalSession(params: {
  stripeCustomerId: string;
}): Promise<{ url: string }> {
  const session = await stripe.billingPortal.sessions.create({
    customer: params.stripeCustomerId,
    return_url: `${process.env.BETTER_AUTH_URL}/settings/billing`,
  });
  return { url: session.url! };
}

export { stripe };
```

Create `apps/web/lib/billing/webhook-handler.ts` (server-only):
```typescript
import 'server-only';
import { stripe } from './stripe';
import { db } from '@/lib/db/client';
import { workspaces } from '@/lib/db/schema/workspaces';
import { eq } from 'drizzle-orm';
import { emitAuditEvent } from '@/lib/auth/audit';
import type Stripe from 'stripe';

export async function handleCheckoutCompleted(session: Stripe.Checkout.Session): Promise<void> {
  const workspaceId = session.metadata?.workspaceId;
  const tier = session.metadata?.tier as 'pro' | 'enterprise';
  const customerId = session.customer as string;
  const subscriptionId = session.subscription as string;

  if (!workspaceId || !tier) {
    console.error('[billing] Missing metadata in checkout session', session.id);
    return;
  }

  await db.update(workspaces)
    .set({
      tier,
      stripeCustomerId: customerId,
      stripeSubscriptionId: subscriptionId,
      subscriptionStatus: 'active',
      subscriptionPeriodEnd: null, // Updated by subscription.updated event
    })
    .where(eq(workspaces.id, parseInt(workspaceId)));

  await emitAuditEvent({
    actorId: 'system',
    event: 'billing_subscription_created',
    metadata: { workspaceId, tier, stripeSubscriptionId: subscriptionId },
  });
}

export async function handleSubscriptionUpdated(subscription: Stripe.Subscription): Promise<void> {
  const workspaceId = subscription.metadata?.workspaceId;
  if (!workspaceId) return;

  const status = subscription.status; // active, past_due, canceled, incomplete
  const periodEnd = subscription.current_period_end
    ? new Date(subscription.current_period_end * 1000)
    : null;

  await db.update(workspaces)
    .set({
      subscriptionStatus: status,
      subscriptionPeriodEnd: periodEnd,
    })
    .where(eq(workspaces.id, parseInt(workspaceId)));

  await emitAuditEvent({
    actorId: 'system',
    event: 'billing_subscription_updated',
    metadata: { workspaceId, status, periodEnd: periodEnd?.toISOString() },
  });
}

export async function handleSubscriptionDeleted(subscription: Stripe.Subscription): Promise<void> {
  const workspaceId = subscription.metadata?.workspaceId;
  if (!workspaceId) return;

  // Downgrade to free tier with read-only access
  await db.update(workspaces)
    .set({
      tier: 'free',
      stripeSubscriptionId: null,
      subscriptionStatus: 'canceled',
      subscriptionPeriodEnd: null,
    })
    .where(eq(workspaces.id, parseInt(workspaceId)));

  await emitAuditEvent({
    actorId: 'system',
    event: 'billing_subscription_canceled',
    metadata: { workspaceId },
  });
}
```

Create `apps/web/app/api/stripe/webhook/route.ts`:
```typescript
import { NextRequest, NextResponse } from 'next/server';
import { stripe } from '@/lib/billing/stripe';
import {
  handleCheckoutCompleted,
  handleSubscriptionUpdated,
  handleSubscriptionDeleted,
} from '@/lib/billing/webhook-handler';

export const runtime = 'nodejs';

export async function POST(request: NextRequest): Promise<NextResponse> {
  const body = await request.text();
  const signature = request.headers.get('stripe-signature')!;

  if (!process.env.STRIPE_WEBHOOK_SECRET) {
    console.error('[billing] STRIPE_WEBHOOK_SECRET not configured');
    return NextResponse.json({ error: 'Webhook secret not configured' }, { status: 500 });
  }

  let event;
  try {
    event = stripe.webhooks.constructEvent(body, signature, process.env.STRIPE_WEBHOOK_SECRET);
  } catch (err) {
    console.error('[billing] Invalid webhook signature:', err instanceof Error ? err.message : err);
    return NextResponse.json({ error: 'Invalid signature' }, { status: 401 });
  }

  try {
    switch (event.type) {
      case 'checkout.session.completed':
        await handleCheckoutCompleted(event.data.object);
        break;
      case 'customer.subscription.updated':
        await handleSubscriptionUpdated(event.data.object);
        break;
      case 'customer.subscription.deleted':
        await handleSubscriptionDeleted(event.data.object);
        break;
      default:
        // Unhandled event type — acknowledged
        console.log(`[billing] Unhandled event type: ${event.type}`);
    }
  } catch (err) {
    console.error('[billing] Webhook handler error:', err);
    return NextResponse.json({ error: 'Handler error' }, { status: 500 });
  }

  return NextResponse.json({ received: true });
}
```

```typescript
import 'server-only';
import { db } from '@/lib/db/client';
import { workspaces } from '@/lib/db/schema/workspaces';
import { eq } from 'drizzle-orm';

export type Tier = 'free' | 'pro' | 'enterprise';
export type SubscriptionStatus = 'active' | 'past_due' | 'canceled' | 'incomplete';

export interface WorkspaceSubscription {
  tier: Tier;
  status: SubscriptionStatus;
  isReadOnly: boolean;
  periodEnd: Date | null;
}

export async function getSubscriptionState(workspaceId: string): Promise<WorkspaceSubscription> {
  const ws = await db.query.workspaces.findFirst({
    where: eq(workspaces.id, parseInt(workspaceId)),
    columns: {
      tier: true,
      subscriptionStatus: true,
      subscriptionPeriodEnd: true,
    },
  });

  const tier = (ws?.tier as Tier) || 'free';
  const status = (ws?.subscriptionStatus as SubscriptionStatus) || 'active';

  // Read-only if subscription is past_due or canceled
  const isReadOnly = status === 'past_due' || status === 'canceled';

  return {
    tier,
    status,
    isReadOnly,
    periodEnd: ws?.subscriptionPeriodEnd || null,
  };
}

export async function isProOrAbove(workspaceId: string): Promise<boolean> {
  const state = await getSubscriptionState(workspaceId);
  return (state.tier === 'pro' || state.tier === 'enterprise') && !state.isReadOnly;
}

export async function isEnterprise(workspaceId: string): Promise<boolean> {
  const state = await getSubscriptionState(workspaceId);
  return state.tier === 'enterprise' && !state.isReadOnly;
}
```

// Feature gate definitions
const FEATURE_GATES = {
  'projects.unlimited': { free: false, pro: true, enterprise: true },
  'members.unlimited': { free: false, pro: true, enterprise: true },
  'savedViews.unlimited': { free: false, pro: true, enterprise: true },

  'cycles.unlimited': { free: false, pro: true, enterprise: true },
  'prioritySupport': { free: false, pro: true, enterprise: true },
  'fileUploads': { free: true, pro: true, enterprise: true }, // 100MB free, 10GB pro+
  'api': { free: false, pro: true, enterprise: true },
  'samlSso': { free: false, pro: false, enterprise: true },
  'auditExport': { free: false, pro: false, enterprise: true },
  'customSla': { free: false, pro: false, enterprise: true },
  'dataExport': { free: true, pro: true, enterprise: true },
} as const;

export type FeatureGate = keyof typeof FEATURE_GATES;

export async function hasFeature(
  workspaceId: string,
  feature: FeatureGate,
): Promise<boolean> {
  const state = await getSubscriptionState(workspaceId);
  if (state.isReadOnly) return false;
  const gate = FEATURE_GATES[feature];
  return gate[state.tier] ?? false;
}

// Limits for free tier

  maxMembers: 5,
  maxProjects: 3,
  maxCycles: 5,
  maxSavedViews: 5,
  fileUploadLimitMB: 100,
} as const;

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Stripe → Webhook endpoint | External payment events cross into our system — must verify HMAC signature |
| Webhook → Postgres | Subscription state writes must be server-only and audited |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-07 | Spoofing | Stripe webhook | mitigate | `stripe.webhooks.constructEvent()` verifies HMAC signature with `STRIPE_WEBHOOK_SECRET`; invalid signatures return 401 |
| T-08-08 | Tampering | Checkout metadata | mitigate | Metadata set server-side only; client cannot modify price ID or tier |
| T-08-09 | Elevation of Privilege | Feature gates | mitigate | `hasFeature()` checks subscription state server-side (import "server-only"); client cannot bypass |
| T-08-10 | Information Disclosure | Stripe keys | mitigate | `STRIPE_SECRET_KEY` is server-only env var; only `NEXT_PUBLIC_STRIPE_PUBLISHABLE_KEY` exposed to client |
| T-08-11 | Denial of Service | Webhook replay | mitigate | Stripe signatures include timestamp; replayed events rejected; idempotent handlers |
</threat_model>
