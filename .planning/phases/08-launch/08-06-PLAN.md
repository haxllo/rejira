Create `apps/web/app/api/stripe/checkout/route.ts`:
```typescript
import { NextRequest, NextResponse } from 'next/server';
import { createCheckoutSession } from '@/lib/billing/stripe';
import { requireAuth } from '@/lib/auth/require-auth';

export const runtime = 'nodejs';

export async function POST(request: NextRequest): Promise<NextResponse> {
  const session = await requireAuth();
  if (!session) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const { workspaceId, tier } = await request.json();

  if (!workspaceId || !tier || !['pro', 'enterprise'].includes(tier)) {
    return NextResponse.json({ error: 'Invalid request' }, { status: 400 });
  }

  // Verify user is workspace admin
  const membership = session.session.activeOrganizationId === workspaceId
    ? session.user // Simplified — real check against memberships table
    : null;
  if (!membership) {
    return NextResponse.json({ error: 'Not workspace admin' }, { status: 403 });
  }

  try {
    const { url } = await createCheckoutSession({
      workspaceId,
      workspaceSlug: workspaceId, // TODO: fetch actual slug
      tier,
      customerEmail: session.user.email,
    });
    return NextResponse.json({ url });
  } catch (err) {
    console.error('[billing] Checkout error:', err);
    return NextResponse.json({ error: 'Failed to create checkout session' }, { status: 500 });
  }
}
```

Create `apps/web/app/api/stripe/portal/route.ts`:
```typescript
import { NextRequest, NextResponse } from 'next/server';
import { createCustomerPortalSession } from '@/lib/billing/stripe';
import { db } from '@/lib/db/client';
import { workspaces } from '@/lib/db/schema/workspaces';
import { eq } from 'drizzle-orm';

export const runtime = 'nodejs';

export async function POST(request: NextRequest): Promise<NextResponse> {
  const { workspaceId } = await request.json();
  
  const ws = await db.query.workspaces.findFirst({
    where: eq(workspaces.id, parseInt(workspaceId)),
    columns: { stripeCustomerId: true },
  });

  if (!ws?.stripeCustomerId) {
    return NextResponse.json({ error: 'No Stripe customer found' }, { status: 404 });
  }

  const { url } = await createCustomerPortalSession({
    stripeCustomerId: ws.stripeCustomerId,
  });

  return NextResponse.json({ url });
}
```
