import type { Metadata } from 'next';

export const metadata: Metadata = {
  title: 'rejira Status',
  description: 'Current system status and incident history for rejira.',
};

const COMPONENTS = [
  { name: 'Web App', status: 'operational' },
  { name: 'API', status: 'operational' },
  { name: 'Database (Supabase)', status: 'operational' },
  { name: 'Authentication (Better Auth)', status: 'operational' },
  { name: 'Realtime (Supabase Realtime)', status: 'operational' },
  { name: 'Storage (Supabase Storage)', status: 'operational' },
  { name: 'Email (Resend)', status: 'operational' },
  { name: 'Billing (Stripe)', status: 'operational' },
];

type Status = 'operational' | 'degraded' | 'outage' | 'maintenance';

const STATUS_COLORS: Record<Status, string> = {
  operational: 'bg-green-500',
  degraded: 'bg-yellow-500',
  outage: 'bg-red-500',
  maintenance: 'bg-blue-500',
};

const STATUS_LABELS: Record<Status, string> = {
  operational: 'Operational',
  degraded: 'Degraded Performance',
  outage: 'Major Outage',
  maintenance: 'Under Maintenance',
};

export default function StatusPage() {
  const allOperational = COMPONENTS.every(c => c.status === 'operational');

  return (
    <div className="max-w-2xl mx-auto px-6 py-16">
      <h1 className="text-2xl font-bold">System Status</h1>
      
      <div className="mt-6 flex items-center gap-2">
        <span className={`w-3 h-3 rounded-full ${allOperational ? 'bg-green-500' : 'bg-yellow-500'}`} />
        <span className="text-lg font-medium">
          {allOperational ? 'All Systems Operational' : 'Some Systems Degraded'}
        </span>
      </div>

      <div className="mt-8 space-y-4">
        {COMPONENTS.map(component => (
          <div key={component.name} className="flex items-center justify-between py-3 border-b border-[var(--color-border)]">
            <span className="text-sm">{component.name}</span>
            <span className="flex items-center gap-2 text-sm text-[var(--color-text-muted)]">
              <span className={`w-2 h-2 rounded-full ${STATUS_COLORS[component.status as Status]}`} />
              {STATUS_LABELS[component.status as Status]}
            </span>
          </div>
        ))}
      </div>

      <div className="mt-12">
        <h2 className="text-lg font-semibold mb-4">Recent Incidents</h2>
        <p className="text-sm text-[var(--color-text-muted)]">No incidents reported in the last 90 days.</p>
      </div>

      <p className="mt-8 text-xs text-[var(--color-text-muted)]">
        Last updated: {new Date().toISOString()}
      </p>
    </div>
  );
}

Create **`apps/web/lib/email/drip.ts`** — Drip email sequence:
```typescript
import 'server-only';
import { sendEmail } from './transport';
import { db } from '@/lib/db/client';

const DRIP_SEQUENCE = [
  {
    delayDays: 0,
    template: 'welcome',
    subject: 'Welcome to rejira! Here\'s how to get started',
  },
  {
    delayDays: 1,
    template: 'first-issue',
    subject: 'Create your first issue in under 30 seconds',
  },
  {
    delayDays: 3,
    template: 'keyboard-shortcuts',
    subject: 'Master rejira with these keyboard shortcuts',
  },
  {
    delayDays: 5,
    template: 'invite-team',
    subject: 'Invite your team and move faster together',
  },
  {
    delayDays: 7,
    template: 'pro-trial',
    subject: 'Try rejira Pro free for 14 days',
  },
  {
    delayDays: 14,
    template: 'check-in',
    subject: 'How\'s rejira working for you?',
  },
];

export async function scheduleDripEmails(userId: string, email: string, name: string): Promise<void> {
  // Store drip schedule in user metadata or a drip_emails table
  // pg_cron job checks for due emails daily and sends them
  
  for (const drip of DRIP_SEQUENCE) {
    const sendAt = new Date(Date.now() + drip.delayDays * 86400000);
    // Insert into drip_schedule table:
    // { userId, email, template, subject, sendAt, sent: false }
  }
  
  // Send welcome email immediately
  await sendEmail({
    to: email,
    subject: DRIP_SEQUENCE[0].subject,
    template: 'welcome',
    data: { name, ctaUrl: `${process.env.BETTER_AUTH_URL}/inbox` },
  }).catch(err => console.error('[drip] Welcome email failed:', err));
}
```
